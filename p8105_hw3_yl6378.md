## Data description and missing values

``` r

  ny_noaa |>
```

## Average maximum temperatures in January and July

``` r
station_temperature =
  noaa |>
  filter(month %in% c(1, 7)) |>
  group_by(id, year, month) |>
  summarize(
    observed_days = sum(!is.na(tmax)),
    mean_tmax = if (all(is.na(tmax))) {
      NA_real_
    } else {
      mean(tmax, na.rm = TRUE)
    },
    .groups = "drop"
  ) |>
  mutate(
    month = factor(
      month,
      levels = c(1, 7),
      labels = c("January", "July")
    )
  )

station_temperature |>
  ggplot(aes(x = year, y = mean_tmax, group = id)) +
  geom_line(alpha = 0.2, linewidth = 0.3, na.rm = TRUE) +
  facet_wrap(~ month) +
  labs(
    title = "Average maximum temperature by station and year",
    x = "Year",
    y = "Mean maximum temperature (°C)"
  ) +
  theme_minimal(base_size = 12)
```

## Temperature relationship and snowfall distributions

``` r
temperature_plot =
  noaa |>
  filter(!is.na(tmin), !is.na(tmax)) |>
  ggplot(aes(x = tmin, y = tmax)) +
  geom_bin_2d(bins = 60) +
  scale_fill_viridis_c(trans = "log10") +
  labs(
    title = "Minimum versus maximum temperature",
    x = "Minimum temperature (°C)",
    y = "Maximum temperature (°C)",
    fill = "Count"
  ) +
  theme_minimal(base_size = 12)

snowfall_plot =
  noaa |>
  filter(snow > 0, snow < 100) |>
  ggplot(aes(x = factor(year), y = snow)) +
  geom_boxplot(outlier.size = 0.3) +
  scale_x_discrete(
    breaks = as.character(seq(1981, 2010, by = 5))
  ) +
  labs(
    title = "Positive snowfall below 100 mm",
    x = "Year",
    y = "Snowfall (mm)"
  ) +
  theme_minimal(base_size = 12)

temperature_plot + snowfall_plot
```

The box plot of snowfall temperature illustrates the distribution of
observed data while avoiding overlapping points that may occur in a full
scatter plot.\
The snowfall box plot compares the median, dispersion, and upper tail of
positive snowfall amounts less than 100 millimeters across multiple
years. Since zero snowfall events and heavy snowfall events of 100
millimeters or more are excluded, this plot does not describe total
annual snowfall or the frequency of snowfall occurrences.
